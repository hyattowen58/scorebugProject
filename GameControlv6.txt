#include <iostream>
using namespace std;

class GameControl
{
private:
    int inning = 1;
    bool topInning = true;
    int ballCount = 0;
    int strikeCount = 0;
    int outCount = 0;
    bool Bases[3] = { false, false, false };
    int topPoints = 0;
    int bottomPoints = 0;

public:
    void recordBall()
    {
        // Adds a ball, every four balls causes a shift to bases by 1
        ballCount++;
        if (ballCount >= 4)
        {
            advanceForceRunners();
            Bases[0] = true;
            ballCount = 0;
            strikeCount = 0;
        }
    }

    void recordStrike()
    {
        // Adds a strike, every three strikes is an out
        strikeCount++;
        if (strikeCount > 2)
        {
            recordOut();
            strikeCount = 0;
        }
    }
    void recordOut()
    {
        // adds an out, three outs causes an inning transition
        outCount++;
        if (outCount > 2)
        {
            triggerInningTransition();
        }
        else
        {
            ballCount = 0;
            strikeCount = 0;
        }
    }
    void recordOut(int basePosition)
    {
        // adds an out that's not caused by the batter, same rules as regular recordOut
        // fixed bug, if runner on base is out, the batters count should not reset
        if (basePosition < 1 || basePosition > 3 || Bases[basePosition - 1] == false)
        {
            return;
        }

        Bases[basePosition - 1] = false;
        outCount++;
        if (outCount > 2)
        {
            triggerInningTransition();
        }
    }
    void handleFoul()
    {
        // adds an strike, unless the batter already has 2 stikes
        if (strikeCount < 2)
        {
            strikeCount++;
        }
    }
    void recordHit(bool single = false, bool doubl = false, bool triple = false, bool homeRun = false)
    {
        //added recordHit to record singles, double, triples, and home runs in their own method without needing to advance forced runners. Simpler for keybinding
        if (single == true)
        {
            advanceAllRunners();
            Bases[0] = true;
        }
        else if (doubl == true)
        {
            advanceAllRunners();
            advanceAllRunners();
            Bases[1] = true;
        }
        else if (triple == true)
        {
            advanceAllRunners();
            advanceAllRunners();
            advanceAllRunners();
            Bases[2] = true;
        }
        else if (homeRun == true)
        {
            advanceAllRunners();
            advanceAllRunners();
            advanceAllRunners();
            if (topInning == true)
            {
                topPoints++;
            }
            else if (topInning == false)
            {
                bottomPoints++;
            }
        }
        ballCount = 0;
        strikeCount = 0;
    }

    void advanceForceRunners()
    {
        // Operates the bases for forced bases movement
        if (Bases[0] == true && Bases[1] == true && Bases[2] == true)   // All bases are loaded
        {
            if (topInning == true)
            {
                topPoints++;
            }
            else if (topInning == false)
            {
                bottomPoints++;
            }
            Bases[0] = false;
        }
        else if (Bases[0] == true && Bases[1] == true)  // Bases 1 and 2 are loaded
        {
            Bases[2] = true;
            Bases[0] = false;
        }
        else if (Bases[0] == true)                                      // Base 1 is loaded
        {
            Bases[1] = true;
            Bases[0] = false;
        }
        else if (Bases[0] == false)                                     // Base 1 is empty
        {
            return;
        }
        else
        {
            cout << "Error, please tell Aldrine";
        }
    }

    void advanceAllRunners()
    {
		//added to simplify hits, and move up every runner not just forced runners. Also allows for multiple bases to be crossed in one hit
        for (int i = 2; i >= 0; i--)
        {
            if (Bases[i] == true)
            {
                if (!(i + 1 > 2))
                {
                    if (Bases[i + 1] == false)
                    {
                        Bases[i + 1] = true;
                        Bases[i] = false;
                    }
                }
                else if (i + 1 > 2)
                {
                    Bases[i] = false;
                    if (topInning == true)
                    {
                        topPoints++;
                    }
                    else if (topInning == false)
                    {
                        bottomPoints++;
                    }
                }
            }
        }
    }
    void resetGame()
    {
        setValues();
    }
    void setBases(bool base1 = false, bool base2 = false, bool base3 = false, bool mistake = false)
    {
        // allows for changes in base positions, also acts as a resetBases() by default
        // new: new default works for runners who cross multiple bases. New option to fix errors
        if (mistake == false)
        {
            Bases[0] = base1;
            Bases[1] = base2;
            Bases[2] = base3;
            ballCount = 0;
            strikeCount = 0;
        }
        else if (mistake == true)
        {
            Bases[0] = base1;
            Bases[1] = base2;
            Bases[2] = base3;
        }

    }
    // private:         // make this public if you want to test this program
    void viewAll()
    {
        cout << "\nInning: " << inning << " Top Inning?: " << topInning << endl;
        cout << "Ball: " << ballCount << " Strike Count: " << strikeCount << " Out Count: " << outCount << endl;
        cout << "Bases: " << Bases[0] << " " << Bases[1] << " " << Bases[2] << endl;
        cout << "Top Inning Points: " << topPoints << " Bottom Inning Points: " << bottomPoints << endl;
    }
    void setValues(int i = 1, bool ti = true, int b = 0, int s = 0, int o = 0, int tp = 0, int bp = 0, bool base1 = false, bool base2 = false, bool base3 = false)
    {
        inning = i;
        topInning = ti;
        ballCount = b;
        strikeCount = s;
        outCount = o;
        topPoints = tp;
        bottomPoints = bp;
        Bases[0] = base1;   Bases[1] = base2;   Bases[2] = base3;
    }
    void triggerInningTransition()
    {
        // clear balls, strikes, outs, bases, and switch to other team
        ballCount = 0;
        strikeCount = 0;
        outCount = 0;
        setBases();
        if (topInning == true)
        {
            topInning = false;
        }
        else if (topInning == false)
        {
            inning++;
            topInning = true;
        }
    }
    
	//getters so that the output can be easily accessed by the main program
    int  getInning() 
    { 
        return inning; 
    }
    bool isTopInning() {
        return topInning; 
    }
    int  getBalls() { 
        return ballCount; 
    }
    int  getStrikes() { 
        return strikeCount; 
    }
    int  getOuts() { 
        return outCount; 
    }
    int  getTopPoints() { 
        return topPoints; 
    }
    int  getBottomPoints() { 
        return bottomPoints; 
    }
    bool getBase(int i) { 
        return Bases[i]; 
    } 
};
